cmake_minimum_required(VERSION 3.22)
project(MiniRemoteClient VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# macOS marks the whole OpenGL 3.2 API we use as deprecated since 10.14; this is
# Apple's documented switch for silencing that, not a way to hide real warnings.
if(APPLE)
  add_compile_definitions(GL_SILENCE_DEPRECATION)
endif()
# A FetchContent subproject that calls export(PACKAGE) registers its build tree
# in the user package registry, which would let a later find_package() resolve
# against another build directory. Keep the registry out of both directions.
set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON)
set(CMAKE_FIND_PACKAGE_NO_PACKAGE_REGISTRY ON)
set(CMAKE_FIND_PACKAGE_NO_SYSTEM_PACKAGE_REGISTRY ON)
option(MRC_BUILD_APP "Build the desktop client" ON)
option(MRC_SSH_INTEGRATION_TESTS "Run a temporary local OpenSSH server for integration tests" OFF)
option(MRC_BUNDLED_VTERM "Build the pinned libvterm source instead of using a system library" OFF)
include(CTest)
find_package(Threads REQUIRED)
include(cmake/PinnedSource.cmake)
include(cmake/Warnings.cmake)
include(cmake/VTerm.cmake)
add_library(mrc_terminal src/terminal.cpp)
target_include_directories(mrc_terminal PUBLIC src)
target_link_libraries(mrc_terminal PUBLIC mrc_vterm)
mrc_target_warnings(mrc_terminal)

if(BUILD_TESTING)
  add_executable(terminal_test tests/terminal_test.cpp)
  target_link_libraries(terminal_test PRIVATE mrc_terminal)
  mrc_target_warnings(terminal_test)
  add_test(NAME terminal COMMAND terminal_test)
endif()

if(MRC_BUILD_APP)
  find_package(OpenGL REQUIRED)
  include(cmake/Glfw.cmake)
  include(cmake/Imgui.cmake)
  include(cmake/Ssh2.cmake)
  add_library(mrc_remote src/remote.cpp)
  target_include_directories(mrc_remote PUBLIC src ${MRC_SSH2_INCLUDE_DIR})
  target_link_libraries(mrc_remote PUBLIC ${MRC_SSH2_TARGET} Threads::Threads)
  add_executable(mini-remote-client src/main.cpp src/app.cpp)
  target_link_libraries(mini-remote-client PRIVATE mrc_terminal mrc_remote imgui::imgui glfw OpenGL::GL)
  if(BUILD_TESTING)
    add_executable(app_test tests/app_test.cpp src/app.cpp)
    target_link_libraries(app_test PRIVATE mrc_terminal mrc_remote imgui::imgui glfw OpenGL::GL)
    add_test(NAME app COMMAND app_test)
    add_executable(remote_test tests/remote_test.cpp)
    target_link_libraries(remote_test PRIVATE mrc_remote)
    mrc_target_warnings(mrc_remote mini-remote-client app_test remote_test)
    if(MRC_SSH_INTEGRATION_TESTS)
      find_package(Python3 REQUIRED COMPONENTS Interpreter)
      add_test(NAME ssh_sftp COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_ssh_test.py $<TARGET_FILE:remote_test>)
      set_tests_properties(ssh_sftp PROPERTIES TIMEOUT 100)
    endif()
  endif()
  install(TARGETS mini-remote-client RUNTIME DESTINATION bin)
endif()
